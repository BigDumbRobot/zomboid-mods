/************************************************************
 * Careers: Mechanic Work Order form recipe
 *
 * A player can craft a blank Mechanic Work Order form and then
 * right-click it to fill in the requested repairs. The alternative
 * is to take a server-crafted form from a mechanic shop's Public
 * business container (which refills itself).
 ************************************************************/

module KIPNCareers
{
    craftRecipe KIPNCareers.CreateMechanicWorkOrder
    {
        timedAction = Making,
        time = 30,
        Tags = InHandCraft,
        category = Miscellaneous,

        inputs
        {
            item 1 [Base.SheetPaper2] flags[ItemCount],
            item 1 tags[base:pen;base:bluepen;base:redpen;base:greenpen] mode:keep flags[Prop1;MayDegradeLight],
            item 1 [Base.Paperclip] flags[ItemCount],
        }

        outputs
        {
            item 1 KIPNCareers.MechanicWorkOrder,
        }
    }
}
